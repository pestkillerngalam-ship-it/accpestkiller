   320→export default function InvoiceDetail({ open, onOpenChange, invoiceId, onRefresh, token }: Props) {
   321→  const [invoice, setInvoice] = useState<Invoice | null>(null);
   322→  const [settings, setSettings] = useState<CompanySettings | null>(null);
   323→  const [loading, setLoading] = useState(false);
   324→  const [activeTab, setActiveTab] = useState('invoice');
   325→  const [previewOpen, setPreviewOpen] = useState(false);
   326→
   327→  useEffect(() => {
   328→    if (open && invoiceId) {
   329→      fetchInvoice();
   330→      fetchSettings();
   331→    }
   332→  }, [open, invoiceId]);
   333→
   334→  const fetchInvoice = async () => {
   335→    if (!invoiceId) return;
   336→    setLoading(true);
   337→    try {
   338→      const res = await fetch(`/api/invoices/${invoiceId}`, {
   339→        headers: { Authorization: `Bearer ${token}` },
   340→      });
   341→      if (res.ok) setInvoice(await res.json());
   342→    } catch (err) {
   343→      console.error(err);
   344→    } finally {
   345→      setLoading(false);
   346→    }
   347→  };
   348→
   349→  const fetchSettings = async () => {
   350→    try {
   351→      const res = await fetch('/api/settings', {
   352→        headers: { Authorization: `Bearer ${token}` },
   353→      });
   354→      if (res.ok) setSettings(await res.json());
   355→    } catch (err) {
   356→      console.error(err);
   357→    }
   358→  };
   359→
   360→  const updateInvoice = async (data: Partial<Invoice>) => {
   361→    if (!invoiceId) return;
   362→    try {
   363→      const res = await fetch(`/api/invoices/${invoiceId}`, {
   364→        method: 'PUT',
   365→        headers: {
   366→          Authorization: `Bearer ${token}`,
   367→          'Content-Type': 'application/json',
   368→        },
   369→        body: JSON.stringify(data),
   370→      });
   371→      if (res.ok) {
   372→        toast.success('Invoice berhasil diperbarui');
   373→        fetchInvoice();
   374→        onRefresh();
   375→      }
   376→    } catch {
   377→      toast.error('Gagal memperbarui invoice');
   378→    }
   379→  };
   380→
   381→  const handleStatusChange = (status: string) => {
   382→    updateInvoice({ status });
   383→  };
   384→
   385→  const handleTaxInvoiceSave = () => {
   386→    if (!invoice) return;
   387→    const isCreated = invoice.taxInvoiceStatus === 'created';
   388→    updateInvoice({
   389→      taxInvoiceStatus: isCreated ? 'not_created' : 'created',
   390→      taxInvoiceDate: isCreated ? null : new Date().toISOString(),
   391→    });
   392→  };
   393→
   394→  const handleImageUpload = (e: React.ChangeEvent<HTMLInputElement>) => {
   395→    const file = e.target.files?.[0];
   396→    if (!file) return;
   397→    const reader = new FileReader();
   398→    reader.onload = () => {
   399→      if (invoice && reader.result) {
   400→        updateInvoice({ taxInvoiceImage: reader.result as string });
   401→      }
   402→    };
   403→    reader.readAsDataURL(file);
   404→  };
   405→
   406→  // Print / PDF functions
   407→  const getPrintHTML = () => {
   408→    if (!invoice) return '';
   409→    return generatePrintHTML(invoice, settings);
   410→  };
   411→
   412→  const handlePrint = () => {
   413→    const html = getPrintHTML();
   414→    const iframe = document.createElement('iframe');
   415→    iframe.style.position = 'fixed';
   416→    iframe.style.left = '-9999px';
   417→    iframe.style.top = '-9999px';
   418→    document.body.appendChild(iframe);
   419→    const doc = iframe.contentDocument || iframe.contentWindow?.document;
   420→    if (!doc) return;
   421→    doc.open();
   422→    doc.write(html);
   423→    doc.close();
   424→    iframe.contentWindow?.focus();
   425→    iframe.contentWindow?.print();
   426→    setTimeout(() => document.body.removeChild(iframe), 5000);
   427→  };
   428→
   429→  const handleDownloadPDF = () => {
   430→    const html = getPrintHTML();
   431→    const iframe = document.createElement('iframe');
   432→    iframe.style.position = 'fixed';
   433→    iframe.style.left = '-9999px';
   434→    iframe.style.top = '-9999px';
   435→    document.body.appendChild(iframe);
   436→    const doc = iframe.contentDocument || iframe.contentWindow?.document;
   437→    if (!doc) return;
   438→    doc.open();
   439→    doc.write(html);
   440→    doc.close();
   441→    // Wait for images to load then trigger print (Save as PDF)
   442→    setTimeout(() => {
   443→      iframe.contentWindow?.print();
   444→      setTimeout(() => document.body.removeChild(iframe), 10000);
   445→    }, 1500);
   446→  };
   447→
   448→  const handleWhatsAppShare = () => {
   449→    if (!invoice) return;
   450→    const msg = `Yth. ${invoice.customer.pic},\n\nBerikut invoice kami:\nNo: ${invoice.invoiceNumber}\nTotal: Rp ${formatCurrency(invoice.total)}\nTanggal: ${formatDate(invoice.issueDate)}\n\nTerima kasih.`;
   451→    const phone = invoice.customer.whatsapp.replace(/[^0-9]/g, '');
   452→    window.open(`https://wa.me/${phone}?text=${encodeURIComponent(msg)}`, '_blank');
   453→  };
   454→
   455→  if (loading || !invoice) {
   456→    return (
   457→      <Dialog open={open} onOpenChange={onOpenChange}>
   458→        <DialogContent className="max-w-3xl">
   459→          <div className="p-8 text-center text-muted-foreground">Memuat...</div>
   460→        </DialogContent>
   461→      </Dialog>
   462→    );
   463→  }
   464→
   465→  const sc = statusConfig[invoice.status] || statusConfig.draft;
   466→
   467→  return (
   468→    <>
   469→      <Dialog open={open} onOpenChange={onOpenChange}>
   470→        <DialogContent className="max-w-4xl max-h-[90vh] overflow-y-auto">
   471→          <DialogHeader>
   472→            <div className="flex items-center justify-between">
   473→              <div className="flex items-center gap-3">
