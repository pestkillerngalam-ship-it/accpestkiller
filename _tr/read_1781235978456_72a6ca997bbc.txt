   288→   288→export default function InvoiceDetail({ open, onOpenChange, invoiceId, onRefresh, token }: Props) {
   289→   289→  const [invoice, setInvoice] = useState<Invoice | null>(null);
   290→   290→  const [settings, setSettings] = useState<CompanySettings | null>(null);
   291→   291→  const [loading, setLoading] = useState(false);
   292→   292→  const [activeTab, setActiveTab] = useState('invoice');
   293→   293→  const [previewOpen, setPreviewOpen] = useState(false);
   294→   294→
   295→   295→  useEffect(() => {
   296→   296→    if (open && invoiceId) {
   297→   297→      fetchInvoice();
   298→   298→      fetchSettings();
   299→   299→    }
   300→   300→  }, [open, invoiceId]);
   301→   301→
   302→   302→  const fetchInvoice = async () => {
   303→   303→    if (!invoiceId) return;
   304→   304→    setLoading(true);
   305→   305→    try {
   306→   306→      const res = await fetch(`/api/invoices/${invoiceId}`, {
   307→   307→        headers: { Authorization: `Bearer ${token}` },
   308→   308→      });
   309→   309→      if (res.ok) setInvoice(await res.json());
   310→   310→    } catch (err) {
   311→   311→      console.error(err);
   312→   312→    } finally {
   313→   313→      setLoading(false);
   314→   314→    }
   315→   315→  };
   316→   316→
   317→   317→  const fetchSettings = async () => {
   318→   318→    try {
   319→   319→      const res = await fetch('/api/settings', {
   320→   320→        headers: { Authorization: `Bearer ${token}` },
   321→   321→      });
   322→   322→      if (res.ok) setSettings(await res.json());
   323→   323→    } catch (err) {
   324→   324→      console.error(err);
   325→   325→    }
   326→   326→  };
   327→   327→
   328→   328→  const updateInvoice = async (data: Partial<Invoice>) => {
   329→   329→    if (!invoiceId) return;

   331→   331→      const res = await fetch(`/api/invoices/${invoiceId}`, {
   332→   332→        method: 'PUT',
   333→   333→        headers: {
   334→   334→          Authorization: `Bearer ${token}`,
   335→   335→          'Content-Type': 'application/json',
   336→   336→        },
   337→   337→        body: JSON.stringify(data),
   338→   338→      });
   339→   339→      if (res.ok) {
   340→   340→        toast.success('Invoice berhasil diperbarui');
   341→   341→        fetchInvoice();
   342→   342→        onRefresh();
   343→   343→      }
   344→   344→    } catch {
   345→   345→      toast.error('Gagal memperbarui invoice');
   346→   346→    }
   347→   347→  };
   348→   348→
   349→   349→  const handleStatusChange = (status: string) => {
   350→   350→    updateInvoice({ status });
   351→   351→  };
   352→   352→
   353→   353→  const handleTaxInvoiceSave = () => {
   354→   354→    if (!invoice) return;
   355→   355→    const isCreated = invoice.taxInvoiceStatus === 'created';
   356→   356→    updateInvoice({
   357→   357→      taxInvoiceStatus: isCreated ? 'not_created' : 'created',
   358→   358→      taxInvoiceDate: isCreated ? null : new Date().toISOString(),
   359→   359→    });
   360→   360→  };
   361→   361→
   362→   362→  const handleImageUpload = (e: React.ChangeEvent<HTMLInputElement>) => {
   363→   363→    const file = e.target.files?.[0];
   364→   364→    if (!file) return;
   365→   365→    const reader = new FileReader();
   366→   366→    reader.onload = () => {
   367→   367→      if (invoice && reader.result) {
   368→   368→        updateInvoice({ taxInvoiceImage: reader.result as string });
   369→   369→      }
   370→   370→    };
   371→   371→    reader.readAsDataURL(file);
   372→   372→  };
   373→   373→
   374→   374→  // Print / PDF functions
   375→   375→  const getPrintHTML = () => {
   376→   376→    if (!invoice) return '';
   377→   377→    return generatePrintHTML(invoice, settings);
   378→   378→  };
   379→   379→
   380→   380→  const handlePrint = () => {
   381→   381→    const html = getPrintHTML();
   382→   382→    const iframe = document.createElement('iframe');
   383→   383→    iframe.style.position = 'fixed';
   384→   384→    iframe.style.left = '-9999px';
   385→   385→    iframe.style.top = '-9999px';
   386→   386→    document.body.appendChild(iframe);
   387→   387→    const doc = iframe.contentDocument || iframe.contentWindow?.document;
   388→   388→    if (!doc) return;
   389→   389→    doc.open();
   390→   390→    doc.write(html);
   391→   391→    doc.close();
   392→   392→    iframe.contentWindow?.focus();
   393→   393→    iframe.contentWindow?.print();
   394→   394→    setTimeout(() => document.body.removeChild(iframe), 5000);
   395→   395→  };
   396→   396→
   397→   397→  const handleDownloadPDF = () => {
   398→   398→    const html = getPrintHTML();
   399→   399→    const iframe = document.createElement('iframe');
   400→   400→    iframe.style.position = 'fixed';
   401→   401→    iframe.style.left = '-9999px';
   402→   402→    iframe.style.top = '-9999px';
   403→   403→    document.body.appendChild(iframe);
   404→   404→    const doc = iframe.contentDocument || iframe.contentWindow?.document;
   405→   405→    if (!doc) return;
   406→   406→    doc.open();
   407→   407→    doc.write(html);
   408→   408→    doc.close();
   409→   409→    // Wait for images to load then trigger print (Save as PDF)
   410→   410→    setTimeout(() => {
   411→   411→      iframe.contentWindow?.print();
   412→   412→      setTimeout(() => document.body.removeChild(iframe), 10000);
   413→   413→    }, 1500);
   414→   414→  };

   416→   416→  const handleWhatsAppShare = () => {

   419→   419→    const phone = invoice.customer.whatsapp.replace(/[^0-9]/g, '');
   420→   420→    window.open(`https://wa.me/${phone}?text=${encodeURIComponent(msg)}`, '_blank');
   421→   421→  };
   422→   422→
   423→   423→  if (loading || !invoice) {
