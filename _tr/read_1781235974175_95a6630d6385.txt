   288→export default function InvoiceDetail({ open, onOpenChange, invoiceId, onRefresh, token }: Props) {
   289→  const [invoice, setInvoice] = useState<Invoice | null>(null);
   290→  const [settings, setSettings] = useState<CompanySettings | null>(null);
   291→  const [loading, setLoading] = useState(false);
   292→  const [activeTab, setActiveTab] = useState('invoice');
   293→  const [previewOpen, setPreviewOpen] = useState(false);
   294→
   295→  useEffect(() => {
   296→    if (open && invoiceId) {
   297→      fetchInvoice();
   298→      fetchSettings();
   299→    }
   300→  }, [open, invoiceId]);
   301→
   302→  const fetchInvoice = async () => {
   303→    if (!invoiceId) return;
   304→    setLoading(true);
   305→    try {
   306→      const res = await fetch(`/api/invoices/${invoiceId}`, {
   307→        headers: { Authorization: `Bearer ${token}` },
   308→      });
   309→      if (res.ok) setInvoice(await res.json());
   310→    } catch (err) {
   311→      console.error(err);
   312→    } finally {
   313→      setLoading(false);
   314→    }
   315→  };
   316→
   317→  const fetchSettings = async () => {
   318→    try {
   319→      const res = await fetch('/api/settings', {
   320→        headers: { Authorization: `Bearer ${token}` },
   321→      });
   322→      if (res.ok) setSettings(await res.json());
   323→    } catch (err) {
   324→      console.error(err);
   325→    }
   326→  };
   327→
   328→  const updateInvoice = async (data: Partial<Invoice>) => {
   329→    if (!invoiceId) return;
   330→    try {
   331→      const res = await fetch(`/api/invoices/${invoiceId}`, {
   332→        method: 'PUT',
   333→        headers: {
   334→          Authorization: `Bearer ${token}`,
   335→          'Content-Type': 'application/json',
   336→        },
   337→        body: JSON.stringify(data),
   338→      });
   339→      if (res.ok) {
   340→        toast.success('Invoice berhasil diperbarui');
   341→        fetchInvoice();
   342→        onRefresh();
   343→      }
   344→    } catch {
   345→      toast.error('Gagal memperbarui invoice');
   346→    }
   347→  };
   348→
   349→  const handleStatusChange = (status: string) => {
   350→    updateInvoice({ status });
   351→  };
   352→
   353→  const handleTaxInvoiceSave = () => {
   354→    if (!invoice) return;
   355→    const isCreated = invoice.taxInvoiceStatus === 'created';
   356→    updateInvoice({
   357→      taxInvoiceStatus: isCreated ? 'not_created' : 'created',
   358→      taxInvoiceDate: isCreated ? null : new Date().toISOString(),
   359→    });
   360→  };
   361→
   362→  const handleImageUpload = (e: React.ChangeEvent<HTMLInputElement>) => {
   363→    const file = e.target.files?.[0];
   364→    if (!file) return;
   365→    const reader = new FileReader();
   366→    reader.onload = () => {
   367→      if (invoice && reader.result) {
   368→        updateInvoice({ taxInvoiceImage: reader.result as string });
   369→      }
   370→    };
   371→    reader.readAsDataURL(file);
   372→  };
   373→
   374→  // Print / PDF functions
   375→  const getPrintHTML = () => {
   376→    if (!invoice) return '';
   377→    return generatePrintHTML(invoice, settings);
   378→  };
   379→
   380→  const handlePrint = () => {
   381→    const html = getPrintHTML();
   382→    const iframe = document.createElement('iframe');
   383→    iframe.style.position = 'fixed';
   384→    iframe.style.left = '-9999px';
   385→    iframe.style.top = '-9999px';
   386→    document.body.appendChild(iframe);
   387→    const doc = iframe.contentDocument || iframe.contentWindow?.document;
   388→    if (!doc) return;
   389→    doc.open();
   390→    doc.write(html);
   391→    doc.close();
   392→    iframe.contentWindow?.focus();
   393→    iframe.contentWindow?.print();
   394→    setTimeout(() => document.body.removeChild(iframe), 5000);
   395→  };
   396→
   397→  const handleDownloadPDF = () => {
   398→    const html = getPrintHTML();
   399→    const iframe = document.createElement('iframe');
   400→    iframe.style.position = 'fixed';
   401→    iframe.style.left = '-9999px';
   402→    iframe.style.top = '-9999px';
   403→    document.body.appendChild(iframe);
   404→    const doc = iframe.contentDocument || iframe.contentWindow?.document;
   405→    if (!doc) return;
   406→    doc.open();
   407→    doc.write(html);
   408→    doc.close();
   409→    // Wait for images to load then trigger print (Save as PDF)
   410→    setTimeout(() => {
   411→      iframe.contentWindow?.print();
   412→      setTimeout(() => document.body.removeChild(iframe), 10000);
   413→    }, 1500);
   414→  };
   415→
   416→  const handleWhatsAppShare = () => {
   417→    if (!invoice) return;
   418→    const msg = `Yth. ${invoice.customer.pic},\n\nBerikut invoice kami:\nNo: ${invoice.invoiceNumber}\nTotal: Rp ${formatCurrency(invoice.total)}\nTanggal: ${formatDate(invoice.issueDate)}\n\nTerima kasih.`;
   419→    const phone = invoice.customer.whatsapp.replace(/[^0-9]/g, '');
   420→    window.open(`https://wa.me/${phone}?text=${encodeURIComponent(msg)}`, '_blank');
   421→  };
   422→
   423→  if (loading || !invoice) {
   424→    return (
   425→      <Dialog open={open} onOpenChange={onOpenChange}>
   426→        <DialogContent className="max-w-3xl">
   427→          <div className="p-8 text-center text-muted-foreground">Memuat...</div>
   428→        </DialogContent>
   429→      </Dialog>
   430→    );
   431→  }
   432→
   433→  const sc = statusConfig[invoice.status] || statusConfig.draft;
   434→
   435→  return (
   436→    <>
   437→      <Dialog open={open} onOpenChange={onOpenChange}>
   438→        <DialogContent className="max-w-4xl max-h-[90vh] overflow-y-auto">
   439→          <DialogHeader>
   440→            <div className="flex items-center justify-between">
   441→              <div className="flex items-center gap-3">
   442→                <DialogTitle className="text-base">{invoice.invoiceNumber}</DialogTitle>
   443→                {invoice.taxInvoiceNumber && (
   444→                  <Badge className="bg-amber-100 text-amber-700 dark:bg-amber-900 dark:text-amber-300 text-xs">
   445→                    FP: {invoice.taxInvoiceNumber}
   446→                  </Badge>
   447→                )}
   448→              </div>
   449→              <div className="flex items-center gap-2">
   450→                <Badge className={sc.className}>{sc.label}</Badge>
   451→                <Select value={invoice.status} onValueChange={handleStatusChange}>
   452→                  <SelectTrigger className="w-32 h-8 text-xs">
   453→                    <SelectValue />
   454→                  </SelectTrigger>
   455→                  <SelectContent>
   456→                    <SelectItem value="draft">Draft</SelectItem>
   457→                    <SelectItem value="unpaid">Belum Lunas</SelectItem>
   458→                    <SelectItem value="paid">Lunas</SelectItem>
   459→                  </SelectContent>
   460→                </Select>
   461→              </div>
   462→            </div>
   463→          </DialogHeader>
   464→
   465→          <Tabs value={activeTab} onValueChange={setActiveTab}>
   466→            <TabsList className="grid w-full grid-cols-3">
   467→              <TabsTrigger value="invoice">Invoice</TabsTrigger>
   468→              <TabsTrigger value="faktur">Faktur Pajak</TabsTrigger>
   469→              <TabsTrigger value="aksi">Aksi</TabsTrigger>
   470→            </TabsList>
   471→
   472→            <TabsContent value="invoice" className="mt-4">
   473→              <div id="invoice-print-area">
   474→                <div className="space-y-6 text-sm">
   475→                  {/* Header */}
   476→                  <div className="flex justify-between items-start gap-4">
   477→                    <div>
   478→                      {settings?.logo && <img src={settings.logo} alt="Logo" className="h-14 mb-2" />}
   479→                      <h2 className="font-bold text-base">{settings?.companyName || 'PT Pest Killer Ngalam'}</h2>
   480→                      <p className="text-muted-foreground text-xs">{settings?.address}</p>
   481→                      <p className="text-muted-foreground text-xs">{settings?.phone} | {settings?.email}</p>
   482→                      {settings?.npwp && <p className="text-muted-foreground text-xs">NPWP: {settings.npwp}</p>}
   483→                    </div>
   484→                    <div className="text-right">
   485→                      <h3 className="font-bold text-lg text-emerald-700">INVOICE</h3>
   486→                      <p className="font-medium">{invoice.invoiceNumber}</p>
   487→                      <p className="text-muted-foreground">Tanggal: {formatDate(invoice.issueDate)}</p>
   488→                      <p className="text-muted-foreground">Jatuh Tempo: {formatDate(invoice.dueDate)}</p>
   489→                    </div>
   490→                  </div>
   491→
   492→                  {/* Tax Invoice Number Badge */}
   493→                  {invoice.taxInvoiceNumber && (
